function fwmTick(el, r){ r.t=0; if(FWM.get(el)!==r || !r.m) return; if(r.st) return fwmRest(el, r); /* FW_SETTLE_MS of play: every part at rest, however far a long round still had to go (an engine without the Web Animations timing calls: here) */
  r.st=fwmClock(r); fwmTime(el, r, true); r.t=setTimeout(()=>fwmTick(el, r), Math.max(0, FW_SETTLE_MS-r.st)); } /* FW_MOVE_MS of play: each loop ends the round it is in */
function fwmTime(el, r, end, x){ try{ (x || el).getAnimations({subtree:true}).forEach(a=>{ if(!r.a.has(a)) fwmA(a, r); else if(end) fwmEnd(a, a.effect.getComputedTiming().localTime); }); }catch(e){} } /* each loop in the spot (x: just one part, whose loop started) timed once */
function fwmStart(e){ const el=e.currentTarget, r=FWM.get(el); if(r && r.m) fwmTime(el, r, false, e.target); } /* a loop made anew while it moves (the spot put back in the page without fwMount): timed at its first frame */
function fwmA(a, r){ r.a.add(a); const e=a.effect; let t=e && FWT.get(a); if(e && !t) FWT.set(a, t=e.getTiming()); /* its timing as the stylesheet wrote it (a loop timed in an earlier move is timed from that again) */
  const d=t ? +t.duration : 0; if(!(d>0) || t.iterations!==Infinity) return; const o=x=>((-t.delay)%x+x)%x;
  if(/^alternate/.test(t.direction)){ const w=((d/2-o(2*d))%d+d)%d; e.updateTiming({delay:w, iterationStart:((o(2*d)+w)/d)%2, iterations:Infinity}); }
  else if(d<=FW_SETTLE_MS/2) e.updateTiming({delay:(d-o(d))%d, iterationStart:0, iterations:Infinity});
  a.currentTime=fwmClock(r); if(r.st) fwmEnd(a, r.st); } /* each loop keeps its own phase (the bundle's offsets: crew never move in step) and waits in its rest pose (the still one) until that phase reaches mid-swing (a back-and-forth loop, whose round starts at an extreme) or its round's start (a bob, a bounce: one round ends where it starts), so nothing jumps from the still pose; at most one round (a long one, as fw-shafts, may not start before the end). Its time is the move's: one made anew carries on where the one it replaces was, and after the step ends where that one ended */
function fwmEnd(a, at){ const e=a.effect, t=e.getTiming(), d=+t.duration, s=t.iterationStart||0, p=s+(at-t.delay)/d; if(!(d>0) || t.iterations!==Infinity || !(p>=s)) return; const i=Math.floor(p);
  e.updateTiming({iterations:(/^alternate/.test(t.direction) ? i+(p-i<.5 ? .5 : 1.5) : i+1)-s}); } /* the round it is in at play time at ends there; a back-and-forth loop stops halfway through a round: mid-swing, where its keyframes (two ends, a symmetric ease) pass within a degree of the rest pose, so taking it off shows no jump (a round's end is an extreme: it snapped). One still waiting (a long round) is left to FW_SETTLE_MS */
